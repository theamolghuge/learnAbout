1. Fundamentals — Types, Variables, Operators & Control Flow

Primitive vs reference types

JavaScript has 7 primitive types and objects for everything else. Primitives are immutable and compared by value. Objects (including arrays and functions) are compared by reference — two variables can point to the same object in memory.

let a = { x: 1 };
let b = a;
b.x = 2;
console.log(a.x); // 2 — same object

let c = 5;
let d = c;
d = 10;
console.log(c);   // 5 — copy of value

const user = { name: "Ada" };

Rule: default to const, use let when you need reassignment, avoid var in new code.

Equality: === vs ==

=== (strict) compares value and type with no conversion. == (loose) coerces types first — know it for interviews, avoid it in production.

// Loose equality coercion rules (interview favorites)
null == undefined;   // true  (special case)
null == 0;           // false
"5" == 5;            // true  (string → number)
"" == 0;             // true
[] == false;         // true  (both → 0)
[] == ![];           // true  (both → 0)

// Strict — use these
0 === false;         // false
NaN === NaN;         // false
Object.is(NaN, NaN); // true
Object.is(+0, -0);  // false (+0 === -0 is true)

Truthy and falsy

Only 8 values are falsy. Everything else — including [], {}, "0", "false" — is truthy.

// Falsy: false, 0, -0, 0n, "", null, undefined, NaN

if ([]) console.log("runs");     // truthy
if ("0") console.log("runs");    // truthy

Important operators

// Nullish coalescing — only null/undefined
const port = config.port ?? 3000;
0 ?? 3000;    // 0
null ?? 3000; // 3000

// Logical OR — any falsy value
0 || 3000;    // 3000

// Optional chaining
user?.profile?.email ?? "none@example.com";
arr?.[0];
callback?.();

// Logical assignment
count ||= 1;   // count = count || 1
count &&= 2;   // count = count && 2
count ??= 1;   // count = count ?? 1

Numbers and strings

0.1 + 0.2 === 0.3;              // false (floating point)
Number.isInteger(3.0);          // true
Number.isSafeInteger(n);        // within ±(2^53 - 1)

parseInt("08", 10);             // always pass radix 10
Number("42");                   // 42
String(42);                     // "42"

`Hello ${name}`;                // template literal
"a".repeat(3);                  // "aaa"

Control flow

else if (score >= 80) grade = "B";
else grade = "C";

// Loops

Lexical scope

Where a function is written determines which variables it can access — not where it is called. Inner functions see outer variables; lookup walks up the chain until global scope.

3. Scope, Hoisting & Execution Context

Scope types

Hoisting

Before execution, the engine registers declarations. var is hoisted and initialized to undefined. Function declarations are hoisted with full body. let/const are hoisted but sit in the Temporal Dead Zone until their line runs.

console.log(a); // undefined — var hoisted
var a = 1;

console.log(b); // ReferenceError — TDZ
let b = 2;

foo();            // works — function declaration hoisted
function foo() { return "hi"; }

bar();            // TypeError — var bar hoisted as undefined
var bar = function () { return "bye"; };

Execution context & call stack

Each time code runs, an execution context is created containing:

  • Variable environment — let/const/var, function declarations, arguments
  • Lexical environment — outer reference for scope chain (closures)
  • this binding — determined by how function is called

The call stack holds active execution contexts (LIFO). Synchronous calls push/pop frames. Too many nested calls → "Maximum call stack exceeded". Async callbacks run later via the event loop — not on the current stack frame.

function outer() {
  const x = 10;
  function inner() {
    console.log(x); // closure — outer lexical env
  }
  inner();
}
outer(); // outer EC → inner EC → pop inner → pop outer

4. Closures

A closure is a function bundled with its lexical environment — it remembers outer variables even after the outer function has finished executing.
function createWallet(initial) {
  let balance = initial;           // private — not accessible outside
  return {
    deposit(n)  { balance += n; return balance; },
    withdraw(n) { balance -= n; return balance; },
    getBalance(){ return balance; }
  };
}
const wallet = createWallet(100);
wallet.deposit(50);   // 150
// wallet.balance — undefined (closure keeps it private)

The var + loop trap (must know)

// BUG: one shared i
for (var i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 0);
}
// 3, 3, 3

// FIX: let creates new binding per iteration
for (let i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 0);

// 0, 1, 2

IIFE — Immediately Invoked Function Expression

(function () {
  const secret = "hidden";
  // secret not in global scope
})();

Used before block scope and modules existed. Still seen in legacy codebases.

Parameters: default, rest, spread

function greet(name = "Guest", greeting = "Hello") {
  return `${greeting}, ${name}`;
}

function sum(first, ...rest) {
  return rest.reduce((a, b) => a + b, first);
}
sum(1, 2, 3, 4); // 10

Higher-order functions

Functions that take or return other functions. Built-ins: map, filter, reduce, sort, forEach.

const nums = [1, 2, 3, 4];

nums.map(n => n * 2);              // [2, 4, 6, 8]
nums.filter(n => n % 2 === 0);     // [2, 4]
nums.reduce((acc, n) => acc + n, 0); // 10
nums.find(n => n > 2);             // 3
nums.some(n => n > 3);             // true
nums.every(n => n > 0);            // true

5. this, Prototypes & Inheritance

How this is bound (priority order)

  1. new binding — new Fn() → new object
  2. Explicit — call / apply / bind
  3. Implicit — obj.method() → obj
  4. Default — plain call → undefined (strict) or globalThis

Arrow functions skip all four — they inherit this from the enclosing scope at write time.

const user = {
  name: "Ada",
  greet() { return this.name; },
  greetArrow: () => this?.name  // lexical — NOT user
};

user.greet();       // "Ada"
const fn = user.greet;
fn();               // undefined (strict) — lost binding

// Fix lost binding
const bound = user.greet.bind(user);
bound();            // "Ada"

call, apply, bind

function introduce(city, country) {
  return `${this.name} from ${city}, ${country}`;
}

introduce.call({ name: "Ada" }, "London", "UK");
introduce.apply({ name: "Ada" }, ["London", "UK"]);

const introFromLondon = introduce.bind({ name: "Ada" }, "London");
introFromLondon("UK"); // "Ada from London, UK"

Prototype chain

Property lookup: own properties first → walk [[Prototype]] chain → null. Methods on Constructor.prototype are shared by all instances — memory efficient.

const arr = [1, 2];
arr.push(3); // not own property — found on Array.prototype
Object.getPrototypeOf(arr) === Array.prototype; // true
Array.prototype.__proto__ === Object.prototype; // true

Prototypal inheritance

Every object has an internal [[Prototype]] link (access via Object.getPrototypeOf(obj)). When you read a property, JS checks the object, then walks up the chain until it finds it or hits null.

function Animal(name) {
  this.name = name;
}
Animal.prototype.speak = function () {
  return `${this.name} makes a sound`;
};

const dog = new Animal("Rex");
dog.speak();  // "Hi, Rex" via prototype

dog.hasOwnProperty("name");   // true  — own property
dog.hasOwnProperty("speak");  // false — on prototype
"name" in dog;                // true  — includes prototype chain

What new does (4 steps)

function myNew(Constructor, ...args) {
  const obj = Object.create(Constructor.prototype);
  const result = Constructor.apply(obj, args);
  return (result !== null && typeof result === "object") ? result : obj;
}

ES6 classes

Syntactic sugar — methods still live on ClassName.prototype. extends sets up the prototype chain; you must call super() in a subclass before using this.

class Animal {
  constructor(name) { this.name = name; }
  speak() { return `${this.name} makes a sound`; }
  static create(name) { return new Animal(name); }
}

class Dog extends Animal {
  constructor(name, breed) {
    super(name);
    this.breed = breed;
  }
  speak() { return `${this.name} barks`; }
}

const d = new Dog("Rex", "Lab");
d instanceof Dog;    // true
d instanceof Animal; // true

Object.create

const proto = { greet() { return "hi"; } };
const obj = Object.create(proto);
obj.greet(); // "hi" — no constructor ran

7. Async JavaScript & Event Loop ⭐

JavaScript is single-threaded

One call stack runs synchronously. Async work (timers, fetch, file I/O) is handled by the browser/Node; callbacks go into queues. The event loop moves them onto the stack when it's free.

Call stack + event loop together

Sync code runs on the call stack until empty. Then the event loop drains microtasks, runs one macrotask, repeats. Async callbacks don't run until the stack is clear — this is why setTimeout(fn, 0) still waits for sync code to finish.

QueueExamples
MicrotasksPromise.then/catch/finally, queueMicrotask, MutationObserver, await continuation
MacrotaskssetTimeout, setInterval, I/O, UI events, requestAnimationFrame

Event loop order

  1. Run all synchronous code on the call stack
  2. Drain the entire microtask queue (Promises, queueMicrotask)
  3. Run one macrotask (setTimeout, I/O, UI events)
  4. Repeat from step 2
console.log("1");

setTimeout(() => console.log("2"), 0);

Promise.resolve().then(() => console.log("3"));

console.log("4");

// Output: 1, 4, 3, 2
// Sync → microtask → macrotask
// async/await adds microtasks
async function demo() {
  console.log("A");
  await null;
  console.log("B");
}
console.log("C");
demo();
console.log("D");
// C, A, D, B

Promises

States: pending → fulfilled or rejected. Once settled, state never changes.

const p = new Promise((resolve, reject) => {
  setTimeout(() => resolve("done"), 1000);
});

p.then(value => console.log(value))
 .catch(err => console.error(err))
 .finally(() => console.log("cleanup"));

// Combinators
Promise.all([p1, p2]);         // all succeed or first reject
Promise.allSettled([p1, p2]);  // wait for all, never rejects
Promise.race([p1, p2]);        // first settle (fulfill or reject)
Promise.any([p1, p2]);         // first fulfill, else AggregateError

async / await

async function fetchUser(id) {
  try {
    const res = await fetch(`/api/users/${id}`);
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    return await res.json();
  } catch (err) {
    console.error("Failed:", err.message);
    throw err;
  }
}

// Parallel (faster)
const [users, posts] = await Promise.all([
  fetchUsers(),
  fetchPosts()
]);

// Sequential (slower — each waits for previous)
const u = await fetchUsers();
const p = await fetchPosts();

Common async patterns

const sleep = ms => new Promise(r => setTimeout(r, ms));

async function retry(fn, attempts = 3, delay = 500) {
  for (let i = 0; i < attempts; i++) {
    try { return await fn(); }
    catch (e) {
      if (i === attempts - 1) throw e;
      await sleep(delay * 2 ** i);
    }
  }
}

AbortController — cancel fetch & async work

const controller = new AbortController();
const { signal } = controller;

fetch("/api/data", { signal })
  .then(r => r.json())
  .catch(err => {
    if (err.name === "AbortError") console.log("Cancelled");
  });

// Cancel from elsewhere
controller.abort();

// Pass signal to multiple operations
const timeout = setTimeout(() => controller.abort(), 5000);

Event bubbling, capturing & delegation

Events propagate in three phases: capturing (window → target), target, bubbling (target → window). Default handlers run in bubbling phase.

// third arg true = capture phase
parent.addEventListener("click", handler, true);

// Stop propagation
event.stopPropagation();
event.preventDefault(); // don't run default (e.g. link navigation)

// Event delegation — one listener on parent
list.addEventListener("click", (e) => {
  const li = e.target.closest("li");
  if (!li || !list.contains(li)) return;
  console.log(li.dataset.id);
});
PhaseDirectionDefault listener?
CaptureRoot → targetOnly if capture: true

13. Memory Management

Shallow vs deep copy (recap)

Shallow copy duplicates top level — nested objects shared. Deep copy duplicates entire tree — use structuredClone or recursive clone with WeakMap for circular refs.

Memory leaks — common causes

  • Event listeners not removed
  • Closures holding references to large objects
  • Global variables accumulating data
  • Detached DOM nodes still referenced in JS
  • setInterval never cleared
  • Unbounded caches (use WeakMap for object keys)

Garbage collection

Mark-and-sweep: engine marks reachable objects from roots (global, call stack, closures), frees unreachable ones. You cannot manually free memory — drop references when done.

Generational GC (V8): new objects in "nursery" (young gen); survivors promoted to old gen. Frequent minor GC, occasional major GC.

14. Performance & Security

Performance optimization

  • Minimize DOM reads/writes — batch, use DocumentFragment
  • Debounce/throttle expensive handlers
  • Lazy load routes and heavy components
  • Avoid layout thrashing (read then write, not interleaved)
  • Use requestAnimationFrame for animations
  • Profile with Chrome DevTools Performance tab — measure before optimizing

Security basics

  • XSS — never insert untrusted HTML; sanitize, use textContent, CSP headers
  • CSRF — tokens, SameSite cookies
  • Prototype pollution — validate object keys from user input
  • Don't expose secrets in client-side JS
  • Validate and sanitize all user input server-side too

Streams (Web Streams API)

const response = await fetch("/large-file");
const reader = response.body.getReader();
while (true) {
  const { done, value } = await reader.read();
  if (done) break;
  processChunk(value); // Uint8Array chunk
}

15. Interview Patterns — Code from Memory

These appear constantly in live coding rounds. Practice writing each blind in 10–15 minutes.

Debounce — wait until user stops

function debounce(fn, delay) {
  let timerId;
  return function (...args) {
    clearTimeout(timerId);
    timerId = setTimeout(() => fn.apply(this, args), delay);
  };
}
// Use: search input, window resize

Throttle — max once per interval

function throttle(fn, limit) {
  let inThrottle = false;
  return function (...args) {
    if (inThrottle) return;
    fn.apply(this, args);
    inThrottle = true;
    setTimeout(() => { inThrottle = false; }, limit);
  };
}
// Use: scroll, mousemove

Curry

function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) {
      return fn.apply(this, args);
    }
    return (...next) => curried(...args, ...next);
  };
}
const add = (a, b, c) => a + b + c;
curry(add)(1)(2)(3); // 6

bind polyfill

Function.prototype.myBind = function (context, ...boundArgs) {
  const fn = this;
  return function (...args) {
    return fn.apply(context, [...boundArgs, ...args]);
  };
};

Promise.all polyfill

function promiseAll(promises) {
  return new Promise((resolve, reject) => {
    const results = [];
    let remaining = promises.length;
    if (remaining === 0) return resolve([]);
    promises.forEach((p, i) => {
      Promise.resolve(p).then(
        val => {
          results[i] = val;
          if (--remaining === 0) resolve(results);
        },
        reject
      );
    });
  });
}

Flatten array

function flatten(arr, depth = Infinity) {
  if (depth === 0) return arr.slice();
  return arr.reduce((acc, val) =>
    acc.concat(Array.isArray(val) ? flatten(val, depth - 1) : val),
  []);
}

Deep clone (with circular references)

function deepClone(value, seen = new WeakMap()) {
  if (value === null || typeof value !== "object") return value;
  if (seen.has(value)) return seen.get(value);

  if (value instanceof Date) return new Date(value);
  if (value instanceof RegExp) return new RegExp(value);

  const copy = Array.isArray(value) ? [] : {};
  seen.set(value, copy);
  for (const key of Reflect.ownKeys(value)) {
    copy[key] = deepClone(value[key], seen);
  }
  return copy;
}

Memoization

function memoize(fn) {
  const cache = new Map();
  return function (...args) {
    const key = JSON.stringify(args);
    if (cache.has(key)) return cache.get(key);
    const result = fn.apply(this, args);
    cache.set(key, result);
    return result;
  };
}
