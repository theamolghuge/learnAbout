5. Generics

function first<T>(arr: T[]): T | undefined {
  return arr[0];
}

interface ApiResponse<T> {
  data: T;
  status: number;
}

function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
  return obj[key];
}

class Store<T> {
  private items: T[] = [];
  add(item: T) { this.items.push(item); }
}

6. Type Narrowing

function printId(id: string | number) {
  if (typeof id === "string") {
    console.log(id.toUpperCase());  // string
  } else {
    console.log(id.toFixed());      // number
  }
}

// Discriminated union
type Shape =
  | { kind: "circle"; radius: number }
  | { kind: "rect"; width: number; height: number };

function area(s: Shape) {
  switch (s.kind) {
    case "circle": return Math.PI * s.radius ** 2;
    case "rect": return s.width * s.height;
  }
}

// Type guard
function isUser(x: unknown): x is User {
  return typeof x === "object" && x !== null && "email" in x;
}

7. Utility Types

UtilityEffect
Partial<T>All props optional
Required<T>All props required
Pick<T, K>Subset of props
Omit<T, K>Exclude props
Record<K, V>Object with key type K, value V
ReturnType<F>Function return type
Parameters<F>Function param tuple
NonNullable<T>Exclude null/undefined
type UpdateUser = Partial<Pick<User, "email" | "role">>;
type UserKeys = keyof User;  // "id" | "email" | "role" | "createdAt"

8. Advanced TypeScript

satisfies & as const

const config = {
  port: 3000,
  env: "production",
} as const;  // readonly, literal types

const routes = {
  home: "/",
  users: "/users",
} satisfies Record<string, string>;  // validate shape, keep inference

Conditional & mapped types

type IsString<T> = T extends string ? true : false;

type ReadonlyFields<T> = {
  readonly [K in keyof T]: T[K];
};

Strict compiler flags (tsconfig)

  • strict: true — enables strictNullChecks, noImplicitAny, etc.
  • noUncheckedIndexedAccess — array access may be undefined
  • exactOptionalPropertyTypes — strict optional handling

9. TypeScript + React

interface ButtonProps {
  label: string;
  onClick: () => void;
  disabled?: boolean;
}

function Button({ label, onClick, disabled = false }: ButtonProps) {
  return <button onClick={onClick} disabled={disabled}>{label}</button>;
}

// Children
interface CardProps {
  title: string;
  children: React.ReactNode;
}

// Events
function Input({ onChange }: { onChange: (e: React.ChangeEvent<HTMLInputElement>) => void }) {}

// useState inference
const [user, setUser] = useState<User | null>(null);

// Generic component
function List<T>({ items, render }: { items: T[]; render: (item: T) => React.ReactNode }) {
  return <ul>{items.map((item, i) => <li key={i}>{render(item)}</li>)}</ul>;
}
